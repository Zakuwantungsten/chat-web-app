```
    ↓
```

**Relationship Properties:**
- Enforce Referential Integrity: YES

```vba
Option Compare Database
Option Explicit

Private Sub Form_Load()
    ' Load combo boxes
    LoadMonthsCombo
    LoadYearsCombo
    LoadTrucksCombo
    LoadRoutesCombo
    
    ' Set default filter to current month
    Me.cmb_Month = Month(Date)
    Me.cmb_Year = Year(Date)
    ApplyDateFilter
End Sub

Private Sub cmb_TruckNo_AfterUpdate()
    ' Auto-fill truck extra fuel
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT ExtraFuel FROM tbl_Trucks WHERE TruckNo = '" & Me.cmb_TruckNo & "'")
    If Not rs.EOF Then
        Me.ExtraFuel = rs!ExtraFuel
    End If
    rs.Close
    Set rs = Nothing
End Sub

Private Sub cmb_Route_AfterUpdate()
    ' Auto-fill total liters based on route
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT TotalLiters, StartPoint, Destination FROM tbl_Routes WHERE RouteID = " & Me.RouteID)
    If Not rs.EOF Then
        Me.TotalLiters = rs!TotalLiters
        Me.StartLocation = rs!StartPoint
        Me.ToLocation = rs!Destination
    End If
    rs.Close
    Set rs = Nothing
    
    ' Calculate initial balance
    CalculateBalance
End Sub

Private Sub CalculateBalance()
    ' Calculate remaining balance
    Dim balance As Long
    balance = Nz(Me.TotalLiters, 0) + Nz(Me.ExtraFuel, 0)
    balance = balance + Nz(Me.MMSAYard, 0) + Nz(Me.TangaYard, 0) + Nz(Me.DarYard, 0)
    balance = balance + Nz(Me.DarGoing, 0) + Nz(Me.MoroGoing, 0) + Nz(Me.MbeyaGoing, 0)
    balance = balance + Nz(Me.TdmGoing, 0) + Nz(Me.ZambiaGoing, 0) + Nz(Me.CongoFuel, 0)
    balance = balance + Nz(Me.ZambiaReturn, 0) + Nz(Me.TundumaReturn, 0)
    balance = balance + Nz(Me.MbeyaReturn, 0) + Nz(Me.MoroReturn, 0)
    balance = balance + Nz(Me.DarReturn, 0) + Nz(Me.TangaReturn, 0)
    
    Me.Balance = balance
End Sub

Private Sub Form_BeforeUpdate(Cancel As Integer)
    ' Set audit fields
    If Me.NewRecord Then
        Me.CreatedBy = GetCurrentUser()
        Me.CreatedDate = Now()
    Else
        Me.ModifiedBy = GetCurrentUser()
        Me.ModifiedDate = Now()
    End If
    
    ' Recalculate balance
    CalculateBalance
End Sub

Private Sub btn_CreateLPO_Click()
    ' Open LPO creation form with this record pre-selected

        MsgBox "Please save the fuel record first.", vbExclamation

Private Sub ApplyDateFilter()
    Dim strFilter As String
    If Not IsNull(Me.cmb_Month) And Not IsNull(Me.cmb_Year) Then
        strFilter = "Month([RecordDate]) = " & Me.cmb_Month & " AND Year([RecordDate]) = " & Me.cmb_Year
        Me.Filter = strFilter
        Me.FilterOn = True
    Else
        Me.FilterOn = False
    End If
End Sub

Private Sub txt_Search_Change()
    ' Dynamic search
    Dim strSearch As String
    If Len(Me.txt_Search) > 0 Then
        strSearch = "TruckNo LIKE '*" & Me.txt_Search & "*' OR " & _
                   "GoingDO LIKE '*" & Me.txt_Search & "*' OR " & _
                   "ReturnDO LIKE '*" & Me.txt_Search & "*'"
        Me.Filter = strSearch
        Me.FilterOn = True
    Else
        Me.FilterOn = False
    End If
End Sub

Private Sub LoadMonthsCombo()

Private Sub LoadYearsCombo()

        yearList = yearList & i & ";"

    Me.cmb_Year.RowSource = Left(yearList, Len(yearList) - 1)
```

**VBA Code for LPO Creation:**

```vba
Option Compare Database
Option Explicit

Private Sub Form_Load()
    ' Generate new LPO number
    Me.txt_LPONumber = GenerateNewLPONumber()
    Me.txt_LPODate = Date
    Me.txt_OrderFor = "TAHMEED"
    
    ' Load stations
    Me.cmb_Station.RowSource = "SELECT StationID, StationName FROM tbl_Stations WHERE IsActive=True ORDER BY StationName"
    
    ' Load checkpoint columns
    LoadCheckpointColumns
    
    ' Load available fuel records
    LoadAvailableFuelRecords
    
    ' Check if opened from fuel record
    If Not IsNull(Me.OpenArgs) Then
        Dim preselectedID As Long
        preselectedID = CLng(Replace(Me.OpenArgs, "FuelRecordID=", ""))
        ' Pre-select this record
        SelectFuelRecord preselectedID
    End If
End Sub

Private Function GenerateNewLPONumber() As String
    Dim rs As DAO.Recordset
    Dim maxLPO As Long
    
    Set rs = CurrentDb.OpenRecordset("SELECT MAX(Val(LPONumber)) AS MaxLPO FROM tbl_LPO")
    If Not rs.EOF Then
        If Not IsNull(rs!MaxLPO) Then
            maxLPO = rs!MaxLPO + 1
        Else
            maxLPO = 2361 ' Starting number
        End If
    End If
    rs.Close
    Set rs = Nothing
    
    GenerateNewLPONumber = CStr(maxLPO)
End Function

Private Sub LoadCheckpointColumns()
    Dim colList As String
    colList = "DarGoing;MoroGoing;MbeyaGoing;TdmGoing;ZambiaGoing;CongoFuel;" & _
              "ZambiaReturn;TundumaReturn;MbeyaReturn;MoroReturn;DarReturn;TangaReturn"
    Me.cmb_CheckpointColumn.RowSource = colList
End Sub

Private Sub LoadAvailableFuelRecords()
    ' Load fuel records with remaining balance
    Dim strSQL As String
    strSQL = "SELECT FuelRecordID, RecordDate, TruckNo, GoingDO, ReturnDO, ToLocation, Balance " & _
             "FROM tbl_FuelRecords " & _
             "WHERE Balance <> 0 AND RecordStatus = 'Active' " & _
             "ORDER BY RecordDate DESC"
    
    Me.lst_AvailableRecords.RowSource = strSQL
    Me.lst_AvailableRecords.ColumnCount = 7
    Me.lst_AvailableRecords.ColumnWidths = "0;1in;1in;0.8in;0.8in;1.2in;0.8in"
End Sub

Private Sub cmb_Station_AfterUpdate()
    ' Get current rate for selected station
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT CurrentRate FROM tbl_Stations WHERE StationID = " & Me.cmb_Station)
    If Not rs.EOF Then
        ' This rate will be used for new detail lines
        Me.Tag = rs!CurrentRate
    End If
    rs.Close
    Set rs = Nothing
End Sub

Private Sub btn_AddToLPO_Click()
    ' Add selected fuel records to LPO details
    Dim varItem As Variant
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim rsFuel As DAO.Recordset
    
    If Me.lst_AvailableRecords.ItemsSelected.Count = 0 Then
        MsgBox "Please select at least one fuel record.", vbExclamation
        Exit Sub
    End If
    
    If IsNull(Me.cmb_Station) Then
        MsgBox "Please select a station first.", vbExclamation
        Exit Sub
    End If
    
    If IsNull(Me.cmb_CheckpointColumn) Then
        MsgBox "Please select which checkpoint column to update.", vbExclamation
        Exit Sub
    End If
    
    Set db = CurrentDb
    
    ' First, save the LPO header if it's new
    If DCount("*", "tbl_LPO", "LPONumber = '" & Me.txt_LPONumber & "'") = 0 Then
        Set rs = db.OpenRecordset("tbl_LPO", dbOpenDynaset)
        rs.AddNew
        rs!LPONumber = Me.txt_LPONumber
        rs!LPODate = Me.txt_LPODate
        rs!StationID = Me.cmb_Station
        rs!OrderFor = Me.txt_OrderFor
        rs!LPOStatus = "Draft"
        rs!CreatedBy = GetCurrentUser()
        rs!CreatedDate = Now()
        rs.Update
        rs.Close
    End If
    
    ' Get LPOID
    Dim lpoID As Long
    lpoID = DLookup("LPOID", "tbl_LPO", "LPONumber = '" & Me.txt_LPONumber & "'")
    
    ' Add detail lines
    Set rs = db.OpenRecordset("tbl_LPODetails", dbOpenDynaset)
    
    For Each varItem In Me.lst_AvailableRecords.ItemsSelected
        Dim fuelRecordID As Long
        fuelRecordID = Me.lst_AvailableRecords.Column(0, varItem)
        
        ' Get fuel record details
        Set rsFuel = db.OpenRecordset("SELECT * FROM tbl_FuelRecords WHERE FuelRecordID = " & fuelRecordID)
        If Not rsFuel.EOF Then
            rs.AddNew
            rs!LPOID = lpoID

Private Function SaveLPO() As Boolean
    On Error GoTo Err_Handler
    
    ' Validate
    If DCount("*", "tbl_LPODetails", "LPOID = (SELECT LPOID FROM tbl_LPO WHERE LPONumber = '" & Me.txt_LPONumber & "')") = 0 Then
        MsgBox "Please add at least one truck to the LPO.", vbExclamation
        SaveLPO = False
        Exit Function
    End If
    
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim rsDetail As DAO.Recordset
    Dim rsFuel As DAO.Recordset
    
    Set db = CurrentDb
    db.BeginTrans
    
    ' Update LPO header with totals
    Set rs = db.OpenRecordset("SELECT * FROM tbl_LPO WHERE LPONumber = '" & Me.txt_LPONumber & "'", dbOpenDynaset)
    If Not rs.EOF Then
        rs.Edit
        rs!TotalLiters = Me.txt_TotalLiters
        rs!TotalAmount = Me.txt_TotalAmount
        rs!LPOStatus = "Sent"
        rs!SentDate = Now()
        rs!SentVia = "System"
        rs!ModifiedBy = GetCurrentUser()
        rs!ModifiedDate = Now()
        rs.Update
    End If
    rs.Close
    
    ' Update fuel records with LPO amounts
    Set rsDetail = db.OpenRecordset("SELECT * FROM tbl_LPODetails WHERE LPOID = " & _
                                     "(SELECT LPOID FROM tbl_LPO WHERE LPONumber = '" & Me.txt_LPONumber & "')")
    
    While Not rsDetail.EOF
        Set rsFuel = db.OpenRecordset("SELECT * FROM tbl_FuelRecords WHERE FuelRecordID = " & rsDetail!FuelRecordID, dbOpenDynaset)
        If Not rsFuel.EOF Then
            rsFuel.Edit
            ' Update the specific checkpoint column with negative value (fuel given out)
            rsFuel(rsDetail!CheckpointColumn) = -rsDetail!Liters
            rsFuel!ModifiedBy = GetCurrentUser()
            rsFuel!ModifiedDate = Now()
            rsFuel.Update
        End If
        rsFuel.Close
        rsDetail.MoveNext
    Wend
    rsDetail.Close
    
    ' Create LPO summary entries
    Call CreateLPOSummaryEntries(Me.txt_LPONumber)
    
    db.CommitTrans
    SaveLPO = True
    Exit Function
    
Err_Handler:
    db.Rollback
    MsgBox "Error saving LPO: " & Err.Description, vbCritical
    SaveLPO = False
End Function
```

```sql
SELECT
```

#### **qry_MonthlyFuelSummary**
```sql
SELECT 
    Format([RecordDate],"mmmm yyyy") AS Month,
    COUNT(*) AS TotalRecords,
    SUM(TotalLiters) AS TotalLitersAllocated,
    SUM(ExtraFuel) AS TotalExtra,
    SUM(Abs([DarGoing] + [MoroGoing] + [MbeyaGoing] + [TdmGoing] + [ZambiaGoing] + 
        [CongoFuel] + [ZambiaReturn] + [TundumaReturn] + [MbeyaReturn] + 
        [MoroReturn] + [DarReturn] + [TangaReturn])) AS TotalLitersUsed
FROM tbl_FuelRecords
GROUP BY Format([RecordDate],"mmmm yyyy"), Year([RecordDate]), Month([RecordDate])
ORDER BY Year([RecordDate]) DESC, Month([RecordDate]) DESC;
```

#### **qry_LPOSummaryByMonth**
```sql
SELECT 
    MonthName,
    Year,
    COUNT(*) AS TotalEntries,
    SUM(Liters) AS TotalLiters,
    SUM(TotalAmount) AS TotalAmount,
    StationName
FROM tbl_LPOSummary
WHERE Status = 'Active'
GROUP BY MonthName, Year, Month, StationName
ORDER BY Year DESC, Month DESC;
```

#### **qry_TruckFuelHistory**
```sql
PARAMETERS [Enter Truck Number] Text;
SELECT 
    RecordDate,
    TruckNo,
    GoingDO,
    ReturnDO,
    ToLocation,
    TotalLiters,
    ExtraFuel,
    (TotalLiters + ExtraFuel) AS TotalAllocated,
    Balance,
    (TotalLiters + ExtraFuel - Balance) AS TotalUsed
FROM tbl_FuelRecords
WHERE TruckNo = [Enter Truck Number]
ORDER BY RecordDate DESC;
```

#### **qry_StationFuelOrders**
```sql
PARAMETERS [Select Station] Long;

    l.StationName,
    d.DONumber,

    d.Destination,
    l.LPOStatus
FROM tbl_LPO l INNER JOIN tbl_LPODetails d ON l.LPOID = d.LPOID
WHERE l.StationID = [Select Station] 
    AND l.LPOStatus <> 'Cancelled'
```

```vba
    Dim i As Integer
```

```vba
Public Sub CalculateCheckpointFuel(fuelRecordID As Long)

    Set rs = CurrentDb.OpenRecordset("SELECT * FROM tbl_FuelRecords WHERE FuelRecordID = " & fuelRecordID, dbOpenDynaset)
    
    If Not rs.EOF Then
        totalLiters = Nz(rs!TotalLiters, 0)
        extraFuel = Nz(rs!ExtraFuel, 0)
        startLocation = Nz(rs!StartLocation, "")
        destination = Nz(rs!ToLocation, "")
        
        rs.Edit
        
        ' GOING PHASE CALCULATIONS
        ' If starting from Tanga
        If UCase(startLocation) = "TANGA" Then
            rs!TangaYard = -100 ' Give 100L to reach Dar
        End If
        
        ' Dar Yard or Going
        If Nz(rs!DarYard, 0) = 0 And Nz(rs!DarGoing, 0) = 0 Then
            ' Check if loaded at Kisarawe
            If InStr(UCase(Nz(rs!FromLocation, "")), "KISARAWE") > 0 Then
                rs!DarYard = -580
            Else
                rs!DarYard = -550
            End If
        End If
        
        ' Mbeya Going
        Dim darFuel As Long
        darFuel = Abs(Nz(rs!DarYard, 0) + Nz(rs!DarGoing, 0))
        If darFuel >= 550 Then
            rs!MbeyaGoing = -450
        Else
            ' If got fuel along the way, give remaining + 450
            rs!MbeyaGoing = -(550 - darFuel + 450)
        End If
        
        ' Zambia Going
        Dim zambiaFuel As Long
        zambiaFuel = (totalLiters + extraFuel) - 900
        
        ' Special destinations
        If InStr(UCase(destination), "LUSAKA") > 0 Then
            rs!ZambiaGoing = -60
        ElseIf InStr(UCase(destination), "LUBUMBASHI") > 0 Then
            rs!ZambiaGoing = -260
        Else
            rs!ZambiaGoing = -zambiaFuel
        End If
        
        ' RETURN PHASE CALCULATIONS
        ' Zambia Return - split between two stations
        rs!ZambiaReturn = -400
        ' Note: In LPO creation, split this as 50L Lake Ndola + 350L Lake Kapiri
        
        ' Tunduma Return
        rs!TundumaReturn = -100
        
        ' Mbeya Return
        rs!MbeyaReturn = -400
        
        ' Moro Return (only for Mombasa destination)
        If InStr(UCase(destination), "MOMBASA") > 0 Then
            rs!MoroReturn = -100
            rs!TangaReturn = -70
        End If
        
        rs!ModifiedBy = GetCurrentUser()
        rs!ModifiedDate = Now()
        rs.Update
    End If
    
    rs.Close
    Set rs = Nothing
End Sub

Public Function GetRecommendedLitersForCheckpoint(fuelRecordID As Long, checkpointColumn As String) As Long
    ' Get recommended liters for a specific checkpoint
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT * FROM tbl_FuelRecords WHERE FuelRecordID = " & fuelRecordID)
    
    If Not rs.EOF Then
        Select Case checkpointColumn
            Case "DarGoing"
                GetRecommendedLitersForCheckpoint = 550 - Abs(Nz(rs!DarYard, 0))
            Case "MbeyaGoing"
                GetRecommendedLitersForCheckpoint = 450
            Case "ZambiaGoing"
                GetRecommendedLitersForCheckpoint = (Nz(rs!TotalLiters, 0) + Nz(rs!ExtraFuel, 0)) - 900
            Case "ZambiaReturn"
                GetRecommendedLitersForCheckpoint = 400
            Case "TundumaReturn"
                GetRecommendedLitersForCheckpoint = 100
            Case "MbeyaReturn"
                GetRecommendedLitersForCheckpoint = 400
            Case "MoroReturn"
                GetRecommendedLitersForCheckpoint = 100
            Case "TangaReturn"
                GetRecommendedLitersForCheckpoint = 70
            Case Else
                GetRecommendedLitersForCheckpoint = 0
        End Select
    End If
    
    rs.Close
    Set rs = Nothing
End Function
```

```vba
Private Sub btn_CancelLPO_Click()
    If MsgBox("Are you sure you want to cancel this LPO? This will reverse fuel record updates.", _
              vbQuestion + vbYesNo) = vbNo Then

    End If
    
    Dim db As DAO.Database
    Dim rs As DAO.Recordset
    Dim rsDetail As DAO.Recordset
    Dim rsFuel As DAO.Recordset

    ' Get LPO details to reverse fuel records
    Set rsDetail = db.OpenRecordset("SELECT * FROM tbl_LPODetails WHERE LPOID = " & Me.LPOID)
    
    While Not rsDetail.EOF
        ' Reverse fuel record update
        Set rsFuel = db.OpenRecordset("SELECT * FROM tbl_FuelRecords WHERE FuelRecordID = " & rsDetail!FuelRecordID, dbOpenDynaset)
        If Not rsFuel.EOF Then
            rsFuel.Edit
            rsFuel(rsDetail!CheckpointColumn) = 0 ' Reset to zero
            rsFuel!ModifiedBy = GetCurrentUser()
            rsFuel!ModifiedDate = Now()
            rsFuel.Update
        End If
        rsFuel.Close
        rsDetail.MoveNext
    Wend
    rsDetail.Close
    
    ' Update LPO status
    Set rs = db.OpenRecordset("SELECT * FROM tbl_LPO WHERE LPOID = " & Me.LPOID, dbOpenDynaset)
    If Not rs.EOF Then
        rs.Edit
        rs!LPOStatus = "Cancelled"
        rs!ModifiedBy = GetCurrentUser()
        rs!ModifiedDate = Now()
        rs.Update
    End If
    rs.Close
    
    ' Cancel LPO summary entries
    db.Execute "UPDATE tbl_LPOSummary SET Status = 'Cancelled' WHERE LPONumber = '" & Me.LPONumber & "'"
    
    db.CommitTrans
    
    MsgBox "LPO cancelled successfully!", vbInformation
    Me.Requery
    Exit Sub
    
Err_Handler:
    db.Rollback
    MsgBox "Error cancelling LPO: " & Err.Description, vbCritical
End Sub
```

```vba
Option Compare Database
Option Explicit

Public Sub ValidateDataIntegrity()
    ' Run comprehensive data validation
    Dim errorCount As Integer
    Dim errorLog As String
    
    errorLog = "DATA INTEGRITY CHECK - " & Format(Now, "yyyy-mm-dd hh:nn:ss") & vbCrLf & String(60, "=") & vbCrLf & vbCrLf
    
    ' 1. Check for orphaned LPO details
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT * FROM tbl_LPODetails WHERE LPOID NOT IN (SELECT LPOID FROM tbl_LPO)")
    If Not rs.EOF Then
        errorCount = rs.RecordCount
        errorLog = errorLog & "ERROR: " & errorCount & " orphaned LPO detail records found." & vbCrLf
    End If
    rs.Close
    
    ' 2. Check for fuel records with invalid balances
    Set rs = CurrentDb.OpenRecordset("SELECT FuelRecordID, TruckNo, Balance FROM tbl_FuelRecords WHERE Balance < 0")
    If Not rs.EOF Then
        errorCount = errorCount + rs.RecordCount
        errorLog = errorLog & "WARNING: " & rs.RecordCount & " fuel records with negative balance." & vbCrLf
        While Not rs.EOF
            errorLog = errorLog & "  - Record ID: " & rs!FuelRecordID & ", Truck: " & rs!TruckNo & ", Balance: " & rs!Balance & vbCrLf
            rs.MoveNext
        Wend
    End If
    rs.Close
    
    ' 3. Check for LPO summary mismatches
    Set rs = CurrentDb.OpenRecordset( _
        "SELECT l.LPONumber FROM tbl_LPO l " & _
        "LEFT JOIN tbl_LPOSummary s ON l.LPONumber = s.LPONumber " & _
        "WHERE l.LPOStatus <> 'Draft' AND s.LPONumber IS NULL")
    If Not rs.EOF Then
        errorCount = errorCount + rs.RecordCount
        errorLog = errorLog & "ERROR: " & rs.RecordCount & " LPOs missing summary entries." & vbCrLf
    End If
    rs.Close
    
    ' 4. Check for duplicate truck numbers
    Set rs = CurrentDb.OpenRecordset( _
        "SELECT TruckNo, COUNT(*) AS DupCount FROM tbl_Trucks " & _
        "GROUP BY TruckNo HAVING COUNT(*) > 1")
    If Not rs.EOF Then
        errorCount = errorCount + rs.RecordCount
        errorLog = errorLog & "ERROR: " & rs.RecordCount & " duplicate truck numbers found." & vbCrLf
    End If
    rs.Close
    
    ' 5. Check for missing truck extra fuel
    Set rs = CurrentDb.OpenRecordset("SELECT TruckNo FROM tbl_FuelRecords WHERE ExtraFuel = 0 OR ExtraFuel IS NULL")
    If Not rs.EOF Then
        errorLog = errorLog & "WARNING: " & rs.RecordCount & " fuel records with no extra fuel assigned." & vbCrLf
    End If
    rs.Close
    
    ' Summary
    errorLog = errorLog & vbCrLf & String(60, "=") & vbCrLf
    If errorCount = 0 Then
        errorLog = errorLog & "RESULT: All integrity checks passed! ✓" & vbCrLf
    Else
        errorLog = errorLog & "RESULT: " & errorCount & " issues found. Please review above." & vbCrLf
    End If
    
    ' Display or save log
    MsgBox errorLog, IIf(errorCount = 0, vbInformation, vbExclamation), "Data Integrity Check"
    
    ' Save to file
    Dim fileNum As Integer
    fileNum = FreeFile
    Open CurrentProject.Path & "\IntegrityCheck_" & Format(Now, "yyyymmdd_hhnnss") & ".txt" For Output As fileNum
    Print #fileNum, errorLog
    Close fileNum
End Sub

Public Sub RecalculateAllBalances()
    ' Recalculate balance for all fuel records
    If MsgBox("This will recalculate balances for all fuel records. Continue?", vbQuestion + vbYesNo) = vbNo Then
        Exit Sub
    End If
    
    Dim rs As DAO.Recordset
    Dim recordCount As Integer
    
    Set rs = CurrentDb.OpenRecordset("SELECT FuelRecordID FROM tbl_FuelRecords", dbOpenDynaset)
    
    While Not rs.EOF
        Dim newBalance As Long
        newBalance = CalculateFuelBalance(rs!FuelRecordID)
        
        CurrentDb.Execute "UPDATE tbl_FuelRecords SET Balance = " & newBalance & _
                         " WHERE FuelRecordID = " & rs!FuelRecordID
        
        recordCount = recordCount + 1
        rs.MoveNext
    Wend
    
    rs.Close
    Set rs = Nothing
    
    MsgBox recordCount & " fuel records updated successfully!", vbInformation
End Sub
```

```vba
Option Explicit

Public Function HasFormAccess(formName As String) As String
    ' Returns: "Full", "ReadOnly", or "None"
    Dim userRole As String
    userRole = TempVars("UserRole")
    
    Dim rs As DAO.Recordset
    Set rs = CurrentDb.OpenRecordset("SELECT " & userRole & " FROM tbl_FormPermissions WHERE FormName = '" & formName & "'")
    
    If Not rs.EOF Then
        HasFormAccess = Nz(rs(userRole), "None")
    Else
        HasFormAccess = "None"
    End If
    
    rs.Close
    Set rs = Nothing
End Function

Public Sub EnforceFormPermissions(frm As Form)

        Case "ReadOnly"
            frm.AllowAdditions = False
            frm.AllowDeletions = False
            frm.AllowEdits = False
        Case "Full"
            ' No restrictions
    End Select
```
